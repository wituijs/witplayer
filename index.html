快速开始 
<!-- 1. 引入播放器 -->
<script src="./witplayer.js"></script>

<!-- 2. 创建容器 -->
<div id="player"></div>

<!-- 3. 初始化播放器 -->
<script>
  const player = new witplayer({
    container: '#player',
    src: 'video.m3u8',
    isM3u8: true
  });

  // API 方法
  player.play();       // 播放
  player.pause();      // 暂停
  player.stop();        // 停止
  player.seek(30);      // 跳转
  player.setVolume(0.5); // 音量
  player.destroy();    // 销毁
</script>
